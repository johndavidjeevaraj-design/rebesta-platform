import 'package:flutter/cupertino.dart';
import 'package:flutter/material.dart';
import 'package:flutter_svg/flutter_svg.dart';

import '../../theme.dart';
import 'package:flutter/material.dart';
import 'package:google_fonts/google_fonts.dart';

class LoginScreen extends StatefulWidget {
  const LoginScreen({super.key});

  @override
  State<LoginScreen> createState() => _LoginScreenState();
}

enum LoginMethod {
  email,
  phone,
}

class _LoginScreenState extends State<LoginScreen> {
  LoginMethod _loginMethod = LoginMethod.email;

  final _emailController = TextEditingController();
  final _passwordController = TextEditingController();
  final _phoneController = TextEditingController();

  final _emailFocus = FocusNode();
  final _passwordFocus = FocusNode();
  final _phoneFocus = FocusNode();

  bool _obscurePassword = true;
  bool _isLoading = false;

  @override
  void dispose() {
    _emailController.dispose();
    _passwordController.dispose();
    _phoneController.dispose();

    _emailFocus.dispose();
    _passwordFocus.dispose();
    _phoneFocus.dispose();

    super.dispose();
  }

  void _changeLoginMethod(LoginMethod method) {
    if (_loginMethod == method) return;

    setState(() {
      _loginMethod = method;
    });

    FocusScope.of(context).unfocus();
  }

  Future<void> _continue() async {
    FocusScope.of(context).unfocus();

    if (_loginMethod == LoginMethod.email) {
      if (_emailController.text.trim().isEmpty) {
        _showError('Please enter your email address');
        return;
      }

      if (_passwordController.text.isEmpty) {
        _showError('Please enter your password');
        return;
      }

      if (!_isValidEmail(_emailController.text.trim())) {
        _showError('Please enter a valid email address');
        return;
      }
    } else {
      if (_phoneController.text.trim().isEmpty) {
        _showError('Please enter your phone number');
        return;
      }

      if (_phoneController.text.trim().length != 10) {
        _showError('Please enter a valid 10-digit phone number');
        return;
      }
    }

    setState(() {
      _isLoading = true;
    });

    // TODO:
    // Connect this to your NestJS / Supabase authentication.
    await Future.delayed(const Duration(milliseconds: 700));

    if (!mounted) return;

    setState(() {
      _isLoading = false;
    });
  }

  bool _isValidEmail(String email) {
    return RegExp(
      r'^[^@\s]+@[^@\s]+\.[^@\s]+$',
    ).hasMatch(email);
  }

  void _showError(String message) {
    ScaffoldMessenger.of(context)
      ..hideCurrentSnackBar()
      ..showSnackBar(
        SnackBar(
          content: Text(message),
          behavior: SnackBarBehavior.floating,
          backgroundColor: AppTheme.errorColor,
          margin: const EdgeInsets.fromLTRB(20, 0, 20, 20),
          shape: RoundedRectangleBorder(
            borderRadius: BorderRadius.circular(14),
          ),
        ),
      );
  }

  @override
  Widget build(BuildContext context) {
    final size = MediaQuery.sizeOf(context);
    final bottomInset = MediaQuery.viewInsetsOf(context).bottom;

    return Scaffold(
      backgroundColor: Colors.transparent,
      resizeToAvoidBottomInset: true,
      body: GestureDetector(
        onTap: () => FocusScope.of(context).unfocus(),
        child: Stack(
          children: [
            // ============================================================
            // FULL SCREEN BACKGROUND
            // ============================================================

            Positioned.fill(
              child: Image.asset(
                'assets/images/login_bg.png',
                fit: BoxFit.cover,
                alignment: Alignment.topCenter,
              ),
            ),

            // ============================================================
            // LOGIN CONTENT
            // ============================================================

            SafeArea(
              child: AnimatedPadding(
                duration: const Duration(milliseconds: 250),
                curve: Curves.easeOut,
                padding: EdgeInsets.only(
                  bottom: bottomInset > 0 ? bottomInset : 0,
                ),
                child: Stack(
                  children: [
                    // ----------------------------------------------------
                    // WHITE SIGN IN PANEL
                    // ----------------------------------------------------

                    Positioned(
                      left: _panelHorizontalMargin(size.width),
                      right: _panelHorizontalMargin(size.width),
                      bottom: _panelBottom(size.height, bottomInset),
                      child: _buildLoginPanel(size),
                    ),

                    // ----------------------------------------------------
                    // BOTTOM FEATURES
                    // ----------------------------------------------------

                    if (bottomInset == 0)
                      Positioned(
                        left: 18,
                        right: 18,
                        bottom: 12,
                        child: _buildFeatureStrip(size),
                      ),
                  ],
                ),
              ),
            ),
          ],
        ),
      ),
    );
  }

  // =========================================================================
  // PANEL POSITION
  // =========================================================================

  double _panelHorizontalMargin(double width) {
    if (width >= 600) {
      return 48;
    }

    return 16;
  }

  double _panelBottom(double height, double keyboardInset) {
    // When keyboard opens, let Flutter move the card above it.
    if (keyboardInset > 0) {
      return 8;
    }

    // This is intentionally lifted slightly upward.
    //
    // This gives us enough room for:
    // Fast Delivery | Best Quality | Secure Payment
    //
    // at the bottom.
    if (height >= 900) {
      return 64;
    }

    if (height >= 800) {
      return 58;
    }

    return 48;
  }

  // =========================================================================
  // LOGIN PANEL
  // =========================================================================

    return Material(
      color: Colors.transparent,
      child: Container(
        width: double.infinity,
        padding: EdgeInsets.fromLTRB(

          10,

        ),

          color: Colors.white.withValues(alpha: 0.96),
          borderRadius: const BorderRadius.vertical(
            top: Radius.circular(32),
            bottom: Radius.circular(32),
          ),
          border: Border.all(
            color: Colors.white.withValues(alpha: 0.9),
            width: 1,
          ),
          boxShadow: [
            BoxShadow(
              color: Colors.black.withValues(alpha: 0.08),
              blurRadius: 28,
              offset: const Offset(0, 10),
            ),
          ],
        ),
        child: Column(
          mainAxisSize: MainAxisSize.min,
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            // ============================================================
            // HANDLE
            // ============================================================

            Center(
              child: Container(
                width: 42,
                height: 5,
                decoration: BoxDecoration(
                  color: const Color(0xFFD7D7D7),
                  borderRadius: BorderRadius.circular(20),
                ),
              ),
            ),

            const SizedBox(height: 14),

            // ============================================================
            // TITLE
            // ============================================================

            Text(
              'Sign in',
              style: GoogleFonts.dmSans(
                fontSize: size.width < 380 ? 25 : 27,
                fontWeight: FontWeight.w700,
                color: const Color(0xFF111827),
                height: 1.05,
              ),
            ),

            const SizedBox(height: 4),

            Text(
              'Order from top restaurants near you.',
              style: GoogleFonts.dmSans(
                fontSize: size.width < 380 ? 13 : 14,
                fontWeight: FontWeight.w400,
                color: const Color(0xFF9CA3AF),
              ),
            ),

            const SizedBox(height: 16),

            // EMAIL / PHONE SWITCHER
            // ============================================================

            _buildLoginSwitcher(size),

            const SizedBox(height: 14),

            // EMAIL MODE
            // ============================================================

            AnimatedSwitcher(
              duration: const Duration(milliseconds: 220),
              switchInCurve: Curves.easeOut,
              switchOutCurve: Curves.easeIn,
              child: isEmail
                  ? _buildEmailForm(size)
                  : _buildPhoneForm(size),
            ),

            const SizedBox(height: 10),

            // ============================================================
            // SOCIAL LOGIN
            // ============================================================

            _buildSocialLogin(size),

            const SizedBox(height: 10),

            // ============================================================
            // CREATE ACCOUNT
            // ============================================================

            _buildCreateAccount(size),
          ],
        ),
      ),
    );
  }

  // =========================================================================
  // SWITCHER
  // =========================================================================

  Widget _buildLoginSwitcher(Size size) {
    final bool isEmail = _loginMethod == LoginMethod.email;

    return Container(
      height: size.width < 380 ? 58 : 62,
      padding: const EdgeInsets.all(4),
      decoration: BoxDecoration(
        color: const Color(0xFFF5F6F8),
        borderRadius: BorderRadius.circular(31),
        border: Border.all(
          color: const Color(0xFFE3E6EA),
        ),
      ),
      child: Row(
        children: [
          Expanded(
            child: _buildSwitchOption(
              selected: isEmail,
              icon: Icons.mail_outline_rounded,
              label: 'Email',
              onTap: () => _changeLoginMethod(LoginMethod.email),
              size: size,
            ),
          ),
          Expanded(
            child: _buildSwitchOption(
              selected: !isEmail,
              icon: Icons.phone_outlined,
              label: 'Phone',
              onTap: () => _changeLoginMethod(LoginMethod.phone),
              size: size,
            ),
          ),
        ],
      ),
    );
  }

  Widget _buildSwitchOption({
    required bool selected,
    required IconData icon,
    required String label,
    required VoidCallback onTap,
    required Size size,
  }) {
    return GestureDetector(
      onTap: onTap,
      behavior: HitTestBehavior.opaque,
      child: AnimatedContainer(
        duration: const Duration(milliseconds: 220),
        curve: Curves.easeOut,
        height: double.infinity,
        decoration: BoxDecoration(
          gradient: selected
              ? const LinearGradient(
                  begin: Alignment.centerLeft,
                  end: Alignment.centerRight,
                  colors: [
                    Color(0xFFFFA51F),
                    Color(0xFFFF3D21),
                  ],
                )
              : null,
          color: selected ? null : Colors.transparent,
          borderRadius: BorderRadius.circular(27),
          boxShadow: selected
              ? [
                  BoxShadow(
                    color: const Color(0xFFFF6B35).withValues(alpha: 0.22),
                    blurRadius: 10,
                    offset: const Offset(0, 4),
                  ),
                ]
              : null,
        ),
        child: Row(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            Icon(
              icon,
              size: size.width < 380 ? 19 : 20,
              color: selected
                  ? Colors.white
                  : const Color(0xFF9CA3AF),
            ),
            const SizedBox(width: 8),
            Text(
              label,
              style: GoogleFonts.dmSans(
                fontSize: size.width < 380 ? 14 : 15,
                fontWeight: FontWeight.w600,
                color: selected
                    ? Colors.white
                    : const Color(0xFF9CA3AF),
              ),
            ),
          ],
        ),
      ),
    );
  }

  // EMAIL FORM

          },

        const SizedBox(height: 12),

        _buildPasswordField(size),

        const SizedBox(height: 7),

        Align(
          alignment: Alignment.centerRight,

            onTap: () {
              // TODO: Forgot password flow.
            },
            child: Text(
              'Forgot password?',
              style: GoogleFonts.dmSans(
                fontSize: size.width < 380 ? 13 : 14,
                fontWeight: FontWeight.w500,
                color: AppTheme.primary,
              ),
            ),
          ),
        ),

        const SizedBox(height: 11),

        _buildPrimaryButton(
          text: 'Continue',
          icon: Icons.arrow_forward_rounded,
          onTap: _continue,
          loading: _isLoading,
          size: size,
        ),
      ],
    );
  }

  // PHONE FORM
  // =========================================================================

          maxLength: 10,

          size: size,

  }

  // =========================================================================
  // INPUT FIELD
  // =========================================================================

  Widget _buildInputField({
    required TextEditingController controller,
    required FocusNode focusNode,
    required String hint,
    required IconData icon,
    required TextInputType keyboardType,
    required TextInputAction textInputAction,
    required Size size,
    required ValueChanged<String> onSubmitted,
    int? maxLength,
  }) {
    return SizedBox(
      height: size.width < 380 ? 54 : 58,
      child: TextField(
        controller: controller,
        focusNode: focusNode,
        keyboardType: keyboardType,
        textInputAction: textInputAction,
        maxLength: maxLength,
        onSubmitted: onSubmitted,
        style: GoogleFonts.dmSans(
          fontSize: 14,
          fontWeight: FontWeight.w500,
          color: const Color(0xFF374151),
        ),
        decoration: InputDecoration(
          counterText: '',
          hintText: hint,
          hintStyle: GoogleFonts.dmSans(
            fontSize: 14,
            color: const Color(0xFFB0B7C3),
          ),
          prefixIcon: Icon(
            icon,
            size: 20,
            color: const Color(0xFFAAB2BE),
          ),
          filled: true,
          fillColor: const Color(0xFFF9FAFB),
          contentPadding: const EdgeInsets.symmetric(
            horizontal: 14,
          ),
          border: OutlineInputBorder(
            borderRadius: BorderRadius.circular(15),
            borderSide: const BorderSide(
              color: Color(0xFFE3E7EC),
            ),
          ),
          enabledBorder: OutlineInputBorder(
            borderRadius: BorderRadius.circular(15),
            borderSide: const BorderSide(
              color: Color(0xFFE3E7EC),
            ),
          ),
          focusedBorder: OutlineInputBorder(
            borderRadius: BorderRadius.circular(15),
            borderSide: const BorderSide(
              color: AppTheme.primary,
              width: 1.4,
            ),
          ),
        ),
      ),
    );
  }

  // =========================================================================
  // PASSWORD
  // =========================================================================

  Widget _buildPasswordField(Size size) {
    return SizedBox(
      height: size.width < 380 ? 54 : 58,
      child: TextField(
        controller: _passwordController,
        focusNode: _passwordFocus,
        obscureText: _obscurePassword,
        textInputAction: TextInputAction.done,
        onSubmitted: (_) => _continue(),
        style: GoogleFonts.dmSans(
          fontSize: 14,
          fontWeight: FontWeight.w500,
          color: const Color(0xFF374151),
        ),
        decoration: InputDecoration(
          hintText: 'Password',
          hintStyle: GoogleFonts.dmSans(
            fontSize: 14,
            color: const Color(0xFFB0B7C3),
          ),
          prefixIcon: const Icon(
            Icons.lock_outline_rounded,
            size: 20,
            color: Color(0xFFAAB2BE),
          ),
          suffixIcon: IconButton(
            onPressed: () {
              setState(() {
                _obscurePassword = !_obscurePassword;
              });
            },
            splashRadius: 20,
            icon: Icon(
              _obscurePassword
                  ? Icons.visibility_off_outlined
                  : Icons.visibility_outlined,
              size: 20,
              color: const Color(0xFFAAB2BE),
            ),
          ),
          filled: true,
          fillColor: const Color(0xFFF9FAFB),
          contentPadding: const EdgeInsets.symmetric(
            horizontal: 14,
          ),
          border: OutlineInputBorder(
            borderRadius: BorderRadius.circular(15),
            borderSide: const BorderSide(
              color: Color(0xFFE3E7EC),
            ),
          ),
          enabledBorder: OutlineInputBorder(
            borderRadius: BorderRadius.circular(15),
            borderSide: const BorderSide(
              color: Color(0xFFE3E7EC),
            ),
          ),
          focusedBorder: OutlineInputBorder(
            borderRadius: BorderRadius.circular(15),
            borderSide: const BorderSide(
              color: AppTheme.primary,
              width: 1.4,
            ),
          ),
        ),
      ),
    );
  }
